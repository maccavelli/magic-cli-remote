name: CI

on:
  push:
    branches: [master]
    tags:
      - "v*"
  pull_request:
  workflow_dispatch:

# One run per ref; a new push to the same PR/branch cancels the older run.
# Tag runs are NEVER cancelled: the release job creates a GitHub Release and
# then uploads assets, so a cancel landing between those two steps leaves a
# published release with missing or partial artifacts.
concurrency:
  group: ci-${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: ${{ github.ref_type != 'tag' }}

# Prefer Node.js 24 LTS (Krypton) for JavaScript-based Actions where supported.
# Do not use Node 20 — it is past LTS for this project.
env:
  FORCE_JAVASCRIPT_ACTIONS_TO_NODE24: true
  NODE_VERSION: "24"
  # Pinned so a Flutter stable bump cannot change analyzer results or APK
  # output without a repo change. Bump deliberately; keep in step with the
  # version developers run locally.
  FLUTTER_VERSION: "3.47.2"

permissions:
  contents: read

jobs:
  # ---------------------------------------------------------------------------
  # Go daemon: vet + race tests + release binaries (tag runs)
  # ---------------------------------------------------------------------------
  go:
    name: Go (test; build on tag)
    runs-on: ubuntu-latest
    timeout-minutes: 25
    # Read-only. The build/<BASE>.<N> ledger is gone (MADR 0005): a release is
    # an immutable strict tag and this job pushes nothing.
    permissions:
      contents: read
    # Version stamped into both binaries; used by the release job artifact names.
    outputs:
      version: ${{ steps.build.outputs.version }}
    # MADR 0118 D4. testexec.SkipIfNoSymlink skips where the symlink privilege
    # is absent, which is right on a developer's unprivileged Windows shell and
    # wrong in CI: a lane that stops verifying three symlink-escape invariants
    # must go red, not quietly green. Inert here — Linux always holds the
    # capability — but set explicitly so a sandbox regression surfaces as a
    # failure rather than a SKIP line nobody reads.
    env:
      MC_REQUIRE_SYMLINK: "1"
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          # Full history + tags for release base (vX.Y.Z) and build/* ledger.
          fetch-depth: 0
          # Needed so GITHUB_TOKEN can push build/* tags on tag runs.
          persist-credentials: true

      - name: Set up Node.js (LTS 24)
        uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020 # v7.0.0
        with:
          node-version: ${{ env.NODE_VERSION }}
          check-latest: true

      - name: Assert Node.js major version
        run: |
          node -v
          major="$(node -p "process.versions.node.split('.')[0]")"
          test "$major" = "24" || { echo "expected Node 24, got $(node -v)" >&2; exit 1; }

      - name: Set up Go
        # v6 runs natively on node24 (v5 still declares node20)
        uses: actions/setup-go@b7ad1dad31e06c5925ef5d2fc7ad053ef454303e # v7.0.0
        with:
          go-version-file: go.mod
          cache: true

      # go vet does not check formatting, so unformatted code used to merge green.
      - name: Gofmt
        run: |
          set -euo pipefail
          unformatted="$(gofmt -l cmd internal)"
          if [ -n "$unformatted" ]; then
            echo "error: gofmt drift — run 'make fmt':" >&2
            echo "$unformatted" >&2
            exit 1
          fi

      - name: Go mod tidy is clean
        run: |
          set -euo pipefail
          cp go.mod /tmp/go.mod.bak && cp go.sum /tmp/go.sum.bak
          go mod tidy
          if ! diff -q go.mod /tmp/go.mod.bak >/dev/null || ! diff -q go.sum /tmp/go.sum.bak >/dev/null; then
            echo "error: go.mod/go.sum are not tidy — run 'make tidy' and commit" >&2
            diff -u /tmp/go.mod.bak go.mod || true
            diff -u /tmp/go.sum.bak go.sum || true
            exit 1
          fi

      - name: Vet
        run: go vet ./...

      # Race-detected run over EVERY package. This replaces the previous
      # `go test ./...` + hand-maintained race allowlist: the allowlist silently
      # excluded new concurrent packages (procutil, admin), and one race pass is
      # measurably faster than a plain pass plus a partial race pass because the
      # test binaries are compiled once instead of twice.
      - name: Test (race, all packages)
        run: go test -race ./...

      # MADR 0116 D20a. The race pass above implies CGO_ENABLED=1, so it links
      # a libc-backed os/user and a net whose resolver selection has a branch
      # the shipped binary does not have. This second pass runs the exact
      # pure-Go stack the linux/amd64 release binary is built with
      # (CGO_ENABLED=0 + netgo,osusergo). Neither pass replaces the other: the
      # first catches data races, the second catches "works under libc, breaks
      # under netgo".
      #
      # Cost: one extra full-suite run on the busiest job, on a free runner.
      - name: Test (cgo-free, all packages)
        run: CGO_ENABLED=0 go test ./...

      # Unit files are shipped to users; a typo'd directive should not reach a
      # release. Shares its implementation with `make preflight` so local and CI
      # cannot drift.
      - name: Validate systemd units
        run: make verify-units

      - name: Verify build-tag policy (Linux netgo; Darwin none)
        run: make verify-build-metadata

      - name: Fetch build version tags
        if: github.ref_type == 'tag'
        run: |
          git fetch origin 'refs/tags/build/*:refs/tags/build/*' --force 2>/dev/null \
            || git fetch origin --tags --force 2>/dev/null \
            || true
          echo "Release tags:"; git tag -l 'v*.*.*' | grep -E '^v[0-9]+\.[0-9]+\.[0-9]+$' | sort -V | tail -5 || true
          echo "Recent build tags:"; git tag -l 'build/*' | sort -V | tail -10 || true

      # Build mcremote and mcrelay for their release platforms, all stamped with
      # one shared VERSION/commit/date.
      #
      # BASE_VERSION is pinned to the tag being built. Without it the allocator
      # resolves BASE from the highest vX.Y.Z tag in the repo, so re-running an
      # older tag (or pushing tags out of order) stamped binaries with a version
      # they were not built from.
      - name: Build mcremote & mcrelay
        id: build
        if: github.ref_type == 'tag'
        env:
          CI: "true"
        run: |
          set -euo pipefail
          BASE="${GITHUB_REF_NAME#v}"
          if [[ ! "$BASE" =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
            echo "error: tag '$GITHUB_REF_NAME' is not vX.Y.Z — cannot derive BASE_VERSION" >&2
            exit 1
          fi

          # Release cross-compile targets — the SAME set for both daemons.
          # build_matrix resolves GOOS/GOARCH; Makefile picks tags per GOOS
          # (MADR 0059 D9: Darwin omits netgo/osusergo).
          PLATFORMS=""
          PLATFORMS="$PLATFORMS linux/amd64"
          PLATFORMS="$PLATFORMS linux/arm64"
          PLATFORMS="$PLATFORMS darwin/arm64"
          PLATFORMS="$PLATFORMS windows/amd64"
          # The Intel Darwin release target is deliberately absent (MADR 0120).
          # windows/arm64 is deliberately absent (MADR 0116 D19). Re-entry is
          # this one line plus a verify-build-metadata c
